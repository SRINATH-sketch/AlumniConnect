#include <stdio.h>
void swap(int *a, int *b) {
int temp = *a;
*a = *b;
*b = temp;
}
int main() {
int n;
printf("Enter Number of Processes: ");
scanf("%d", &n);
int b[n], p[n], index[n];
for (int i = 0; i < n; i++) {
printf("Enter Burst Time and Priority Value for Process %d: ", i + 1);
scanf("%d %d", &b[i], &p[i]);
index[i] = i + 1;
}
for (int i = 0; i < n; i++) {
int highest_priority = p[i];
int highest_index = i;
for (int j = i + 1; j < n; j++) {
if (p[j] > highest_priority) {
highest_priority = p[j];
highest_index = j;
}
}swap(&p[i], &p[highest_index]);swap(&b[i], &b[highest_index]);
swap(&index[i], &index[highest_index]);
}
int t = 0;
printf("Order of Process Execution is:\n");
for (int i = 0; i < n; i++) {
printf("P%d is executed from %d to %d\n", index[i], t, t + b[i]);
t += b[i];
}
printf("\nProcess Id Burst Time Wait Time Turnaround Time\n");
int wait_time = 0;
for (int i = 0; i < n; i++) {
printf("P%d %d %d %d\n", index[i], b[i], wait_time, wait_time + b[i]);
wait_time += b[i];
}
return 0;
}
